    ۱۲. ګرامر (EBNF)#

    ګرامر د انګلیسي سند ۱۲ برخه کې دی (docs/language.md)؛ د کلیدي کلمو پښتو بڼې پورته جدول ورکوي. لنډیز: امرونه د کرښې په پای، بلاکونه د INDENT/DEDENT ترمنځ، افادې د Pratt الګوریتم په لومړیتوبونو (۵ برخه)، او د تکرار سرلیک درې برخې په کامو بېلې دي.

    د نښو، ټولګو، جوړښتونو، ګډونونو، برسېرونو او «سره» قاعدې، د پښتو کلیدي کلمو سره:

    declaration    = { annotation } ( function | native_func | struct | mixin | interface | enum | module_decl )
               | import | statement ;
annotation     = "@" IDENTIFIER [ "(" STRING ")" ] NEWLINE ;   (* @ازموینه، @ازموینه("جمع سمه ده")، @پېښه ګر؛
                                                                   کوم اعلان یې نښه کولای شي وروسته کتل کېږي (S085) *)
module_decl    = "ټولګی" IDENTIFIER INDENT { member } DEDENT ;   (* یوازې د فایل په سر کچه *)
member         = { annotation } ( function | native_func | struct | mixin | interface | enum ) | var_decl end ;
qualified_name = IDENTIFIER { "." IDENTIFIER } ;   (* هندسه.ټکی، ریاضي.جذر، ګرافیک.رنګونه.سور *)
type           = ... | qualified_name [ "<" type { "," type } ">" ] | ... ;
struct         = "جوړښت" IDENTIFIER [ "<" IDENTIFIER { "," IDENTIFIER } ">" ]
                 ( ":" qualified_name { "," qualified_name }      (* برسېرونه او ګډونونه ... *)
                   ( INDENT { struct_member } DEDENT | end )      (* ... چې له هغوی وروسته تنه پرېښودل کېدای شي *)
                 | INDENT { struct_member } DEDENT ) ;
mixin          = "ګډون" IDENTIFIER [ ":" qualified_name { "," qualified_name } ]   (* یوازې برسېرونه *)
                 INDENT { struct_member } DEDENT ;               (* جوړوونکی نه *)
struct_member  = IDENTIFIER ":" type [ "," ]                       (* برخه *)
               | IDENTIFIER "(" ")" ":" function_type [ "," ]      (* بلل کېدونکې برخه (closure) *)
               | { annotation } "دنده" IDENTIFIER "(" [ parameter { "," parameter } ] ")" [ ":" type ] block  (* دنده *)
               | IDENTIFIER "(" [ parameter { "," parameter } ] ")" block ;  (* جوړوونکې دنده: د جوړښت په نوم *)
interface      = "برسېر" IDENTIFIER INDENT { interface_member } DEDENT ;
interface_member = "دنده" IDENTIFIER "(" [ parameter { "," parameter } ] ")" [ ":" type ] end
               | IDENTIFIER "(" ")" ":" function_type [ "," ] ;    (* بلل کېدونکی غړی *)
lambda         = "دنده" "(" [ parameter { "," parameter } ] ")" [ ":" type ]
                 ( simple_stmt | expression (* د راګرځېدو ډول سره *) | block (* له قوسونو بهر *) ) ;
with_stmt      = IDENTIFIER "سره" [ ":" type ] "=" expression block   (* پښتو: سره له نوم وروسته *)
               | expression "سره" block            (* پښتو: بې نومه، سره کرښه پای ته رسوي *)
               | "سره" with_head block ;            (* انګلیسي ترتیب: P018، خو بیا هم پوهېږي *)
with_head      = [ IDENTIFIER [ ":" type ] "=" ] expression ;
